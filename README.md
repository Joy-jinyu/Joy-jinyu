<h3>🧪 &nbsp;What my agent is probably streaming</h3>

```ts
// 把 Agent 的一次回答，变成人看得懂、跟得上、接得住的界面
for await (const event of agent.run(task)) {
  switch (event.type) {
    case 'reasoning': ui.think(event.text); break             // 想的过程能看，但收着
    case 'tool_call': ui.step(event.tool, event.args); break  // 调了什么，一眼对上
    case 'delta':     ui.stream(event.text); break            // 停在你正在读的地方
    case 'artifact':  ui.deliver(event.artifact); break       // 能预览、能下载、能分享
    case 'error':     ui.recover(event); break                // 先重连，再给出路
  }
}
// AI 写得出界面，写不出分寸。分寸是我的活。
```
